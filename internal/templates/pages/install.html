Setup Repository

Option 1: Modern Format (.sources) - Ubuntu 24.04 LTS and newer

# Download and add the GPG key
curl {{ .RepositoryURL }}/pubkey.gpg | sudo gpg --dearmor -o /usr/share/keyrings/repository.gpg

# Create repository source file
sudo tee /etc/apt/sources.list.d/repository.sources > /dev/null <<'EOF'
Types: deb
URIs: {{ .RepositoryURL }}
Suites: {{ .Suite }}
Components: main
Signed-By: /usr/share/keyrings/repository.gpg
EOF

# Update package lists
sudo apt update

# Install packages
sudo apt install package-name

Option 2: Traditional Format (.list) - All Debian/Ubuntu versions

# Download and add the GPG key
curl {{ .RepositoryURL }}/pubkey.gpg | sudo gpg --dearmor -o /usr/share/keyrings/repository.gpg

# Add the repository
echo "deb [signed-by=/usr/share/keyrings/repository.gpg] {{ .RepositoryURL }} {{ .Suite }} main" | \
  sudo tee /etc/apt/sources.list.d/repository.list

# Update package lists
sudo apt update

# Install packages
sudo apt install package-name

Verify Key

# Check the GPG key details
gpg --show-keys /usr/share/keyrings/repository.gpg

# Verify package signatures
apt-cache policy package-name

Configure Credentials

If the repository requires authentication, create an apt credentials file:

# Create auth config directory if it doesn't exist
sudo mkdir -p /etc/apt/auth.conf.d

# Add your credentials (using the repository domain from above)
sudo tee /etc/apt/auth.conf.d/repository.conf > /dev/null <<'EOF'
machine debs.example.com
login your-username
password your-password
EOF

# Restrict permissions (required!)
sudo chmod 600 /etc/apt/auth.conf.d/repository.conf

# Test the connection
sudo apt update

Note: The auth.conf.d file permissions must be 600 (readable only by root) for apt to use the credentials.
